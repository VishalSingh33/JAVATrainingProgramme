1. Create

db.books.insertOne({
    "title": "Spider Man",

    "pages": 250,

});

2. Read
Find All Documents:
db.books.find().pretty();

Find Document by Title:
db.books.find({ "title": "Spider Man" }).pretty();

Find Documents with Rating Greater than 4.5:
db.books.find({ "rating": { $gt: 4.5 } }).pretty();


Find Document by Title and Project Specific Fields:
db.books.find(
    { "title": "Spider Man" },
    { "title": 1, "author": 1, "_id": 0 }
).pretty();


3. Update
Update a Single Document by Title:
db.books.updateOne(
    { "title": "Spider Man" },
    { $set: { "rating": 4.3 } }
);


Update Multiple Documents:
db.books.updateMany(
    { "genre": "Fiction" },
    { $set: { "rating": 4.7 } }
);


Add a New Field to All Documents:
db.books.updateMany(
    {},
    { $set: { "available": true } }
);

Replace a Document:
db.books.replaceOne(
    { "title": "Spider Man" },
    { "title": "Spider Man", "author": "Haanah", "pages": 250, "genre": ["Fiction"], "rating": 4.5, "available": true }
);


4. Delete
Delete a Single Document by Title:
db.books.deleteOne({ "title": "Spider Man" });

Delete Multiple Documents:
db.books.deleteMany({ "rating": { $lt: 4.5 } });


Drop the Collection:
db.books.drop();

5. Aggregation:
For all 
db.books.aggregate([
    {
        $lookup: {
            from: "authors",            // Collection to join
            localField: "authorId",     // Field from `books`
            foreignField: "_id",        // Field from `authors`
            as: "authorDetails"         // Alias for the resulting array
        }
    },
    {
        $unwind: "$authorDetails"       // Deconstruct the array to output a document for each element
    },
    {
        $project: {                     // Specify the fields to include or exclude
            title: 1,
            pages: 1,
            genre: 1,
            rating: 1,
            "authorDetails.name": 1,
            "authorDetails.country": 1,
            "authorDetails.birthYear": 1
        }
    }
]).pretty();


For particular data
db.books.aggregate([
    {
        $lookup: {
            from: "authors",
            localField: "authorId",
            foreignField: "_id",
            as: "authorDetails"
        }
    },
    {
        $unwind: "$authorDetails"
    },
    {
        $match: {
            "title": "Iron Man",          // Filter by title
            "authorDetails.name": "Aditi"  // Filter by author name
        }
    },
    {
        $project: {
            title: 1,
            pages: 1,
            genre: 1,
            rating: 1,
            "authorDetails.name": 1,
            "authorDetails.country": 1,
            "authorDetails.birthYear": 1
        }
    }
]).pretty();
